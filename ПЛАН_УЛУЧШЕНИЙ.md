### 7. Настройка CI/CD

**Создать:** `.github/workflows/ci.yml`

```yaml
name: CI/CD

on:
  push:
    branches: [main]
  pull_request:
    branches: [main]

jobs:
  # Проверка форматирования
  lint:
    name: Lint & Format Check
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4
        
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'npm'
          
      - name: Install dependencies
        run: npm ci
        
      - name: Check formatting
        run: npm run prettier:check

  # Сборка проекта
  build:
    name: Build
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4
        
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'npm'
          
      - name: Install dependencies
        run: npm ci
        
      - name: Build
        run: npm run build
        
      - name: Upload build artifacts
        uses: actions/upload-artifact@v4
        with:
          name: build
          path: build/
          retention-days: 7

  # Деплой на GitHub Pages
  deploy:
    name: Deploy to GitHub Pages
    runs-on: ubuntu-latest
    needs: [lint, build]
    if: github.ref == 'refs/heads/main' && github.event_name == 'push'
    permissions:
      contents: write
    steps:
      - name: Checkout
        uses: actions/checkout@v4
        
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'
          cache: 'npm'
          
      - name: Install dependencies
        run: npm ci
        
      - name: Deploy
        run: npm run deploy
        env:
          GIT_USER: ${{ github.actor }}
          GIT_PASS: ${{ secrets.GITHUB_TOKEN }}
```

### 9. Оптимизация изображений

**Скрипт оптимизации:**

```bash
#!/bin/bash
# optimize-images.sh

echo "🖼️  Оптимизация изображений..."

# Установка инструментов
npm install -g sharp-cli

# PNG оптимизация
find static/img -name "*.png" -exec sharp -i {} -o {}.optimized \;
find static/img -name "*.png.optimized" -exec sh -c 'mv "$1" "${1%.optimized}"' _ {} \;

# JPG оптимизация
find static/img -name "*.jpg" -o -name "*.jpeg" -exec sharp -i {} -o {}.optimized -q 85 \;
find static/img -name "*.jpg.optimized" -o -name "*.jpeg.optimized" -exec sh -c 'mv "$1" "${1%.optimized}"' _ {} \;

echo "✅ Оптимизация завершена"
```

**Добавить в package.json:**

```json
{
  "scripts": {
    "optimize:images": "bash scripts/optimize-images.sh"
  }
}
```
